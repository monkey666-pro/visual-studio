​			{

​				[变量类型对应的变量名(b)]=new Dictionary <数据类型，数据类型>()

​				{

​					可以输入初次设置的变量名给不同的值（独立的）//a1

​				}

​			}

```
			Console.WriteLine(变量名[a])		//输出初次设置的集合
```

###### 4、list用法，结构

​		概念：**强类型**的泛型集合

​				强类型：变量必须声明类型，弱类型反之

​		用法：list <string> A = new <string>(){}

​			add(A.add(内容))

​				在末尾增加数据

​			addrange(A.addrange([内容]))

​				在增加多处数据

​			insert(A.insert(目标位置(索引),内容))

​				在目标索引出增加数据

​			Remove (A.remove(数据数据))

​				删除集合中所填的数据

​			RemoveAT(A.removeat(索引))

​				删除指定索引的数据

​			RemoveRange(A.Removerange(索引，数字))

​				从所填索引数开始，删除所填数字的个数

​			Clear(A.clear())

​				删除集合中所有的数据

​			constains(A.constains(数据))

​				判断该集合中是否有所填数据（输出为布尔型）

​			indexOf(A.indexOf(集合中的数据))

​				得到所填数据第一次的索引

​			LastindexOf(A.LastindexOf(集合中的数据))

​				得到所填数据最后一次的索引

​			GetRange(A.GetRange(索引，个数))

​				得到A集合中从所填索引开始到所填个数的值（所得的为集合）

​			Reverse(A.Reverse())

​				将集合中的数据反过来
